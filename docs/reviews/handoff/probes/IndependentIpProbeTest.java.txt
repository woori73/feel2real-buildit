package f2r.tutor_solution.private_tutor_solution_back.domain.admissions.controller;

import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.Mockito.*;

import jakarta.servlet.http.HttpServletRequest;
import org.junit.jupiter.api.Test;

/** Claude 독립 검수용: 프록시 IP 위조 벡터(MCP가 재사용하는 resolveClientIp). */
class IndependentIpProbeTest {
  private String resolve(String trusted, String remote, String xff) {
    var l = new AdmissionsRateLimiter(10, 60, 1000, trusted, () -> 0L);
    var r = mock(HttpServletRequest.class);
    when(r.getRemoteAddr()).thenReturn(remote);
    when(r.getHeader("X-Forwarded-For")).thenReturn(xff);
    return l.resolveClientIp(r);
  }

  @Test
  void spoofVectors() {
    assertEquals("203.0.113.5", resolve("10.0.0.0/8", "203.0.113.5", "1.1.1.1"), "신뢰하지 않는 상대의 XFF는 무시");
    assertEquals("8.8.8.8", resolve("10.0.0.0/8", "10.1.1.1", "9.9.9.9, 8.8.8.8"), "앞에 덧붙인 위조값 무시");
    assertEquals("8.8.8.8", resolve("10.0.0.0/8", "10.1.1.1", "8.8.8.8, 10.2.2.2"), "다단 프록시: 오른쪽부터 비신뢰 첫 값");
    assertEquals("10.1.1.1", resolve("10.0.0.0/8", "10.1.1.1", "abc"), "형식 오류는 직접 주소");
    assertEquals("10.1.1.1", resolve("10.0.0.0/8", "10.1.1.1", "8.8.8.8,"), "빈 항목은 직접 주소");
    assertEquals("10.1.1.1", resolve("10.0.0.0/8", "10.1.1.1", null));
    // 운영 기본값(trusted-proxies 비어 있음): ALB 뒤에서는 모든 요청이 ALB 주소 하나로 집계됨
    assertEquals("10.1.1.1", resolve("", "10.1.1.1", "8.8.8.8"), "기본값에서는 XFF를 쓰지 않는다 = 사용자 구분 불가");
    assertEquals("10.1.1.1", resolve("0.0.0.0/0", "10.1.1.1", "8.8.8.8"), "과도하게 넓은 목록은 무시되어야 함");
  }
}
