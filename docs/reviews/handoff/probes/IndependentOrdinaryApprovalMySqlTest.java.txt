package f2r.tutor_solution.private_tutor_solution_back.domain.easy;

import static org.assertj.core.api.Assertions.assertThat;
import static org.mockito.ArgumentMatchers.any;
import static org.mockito.ArgumentMatchers.eq;
import static org.mockito.Mockito.when;

import f2r.tutor_solution.private_tutor_solution_back.config.security.JwtTokenProvider;
import f2r.tutor_solution.private_tutor_solution_back.domain.easy.entity.EasyBusiness;
import f2r.tutor_solution.private_tutor_solution_back.domain.easy.repository.EasyBusinessRepository;
import f2r.tutor_solution.private_tutor_solution_back.domain.easy.repository.EasyTransactionRepository;
import f2r.tutor_solution.private_tutor_solution_back.domain.easy.service.EasyTutorAccessService;
import f2r.tutor_solution.private_tutor_solution_back.domain.easy.service.EasyTutorImportService;
import java.io.ByteArrayInputStream;
import java.nio.charset.StandardCharsets;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.test.context.bean.override.mockito.MockitoBean;
import org.springframework.transaction.annotation.Transactional;

/** Claude 독립 검수: 가드 수정 후에도 일반(강사·비서) CSV 업로드→사용자 확정 정상 흐름이 동작하는가. */
@SpringBootTest
@Transactional
class IndependentOrdinaryApprovalMySqlTest {
  @Autowired EasyTutorImportService tutorImport;
  @Autowired EasyBusinessRepository businesses;
  @Autowired EasyTransactionRepository transactions;
  @MockitoBean EasyTutorAccessService access;

  @Test
  void ordinaryTutorCsvStillConfirmsOnceAndOnlyOnce() throws Exception {
    EasyBusiness a = businesses.findAll().stream().filter(EasyBusiness::isActive).findFirst().orElseThrow();
    when(access.requireAllowedBusiness(any(), eq(a.getBusinessId()))).thenReturn(a);
    var tutor = new JwtTokenProvider.AuthenticatedPrincipal(10L, "tutor-a", "TUTOR");
    long before = transactions.findAll().stream().filter(t -> t.getBusiness().getBusinessId().equals(a.getBusinessId())).count();
    String csv = "거래일,구분,금액,거래처,적요,결제수단,원본계정,증빙유형\n2026-10-22,지출,5000,정상검증처,일반 CSV 확정 회귀,계좌이체,검증계좌,NONE\n";
    var staged = tutorImport.uploadCsv(tutor, a.getBusinessId(), "ordinary.csv", new ByteArrayInputStream(csv.getBytes(StandardCharsets.UTF_8)));
    long afterStage = transactions.findAll().stream().filter(t -> t.getBusiness().getBusinessId().equals(a.getBusinessId())).count();
    assertThat(afterStage).as("승인 전 stage는 장부에 쓰지 않는다").isEqualTo(before);
    var confirmed = tutorImport.confirm(tutor, a.getBusinessId(), staged.batchId(), null);
    long afterConfirm = transactions.findAll().stream().filter(t -> t.getBusiness().getBusinessId().equals(a.getBusinessId())).count();
    System.out.println("PROBE ordinary status=" + confirmed.status() + " confirmedRows=" + confirmed.confirmedRows() + " ledger " + before + "->" + afterConfirm);
    assertThat(afterConfirm).isEqualTo(before + 1);
  }
}
