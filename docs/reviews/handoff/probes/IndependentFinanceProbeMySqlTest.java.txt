package f2r.tutor_solution.private_tutor_solution_back.domain.easy;

import static org.assertj.core.api.Assertions.assertThat;
import static org.assertj.core.api.Assertions.assertThatThrownBy;
import static org.mockito.ArgumentMatchers.any;
import static org.mockito.ArgumentMatchers.eq;
import static org.mockito.Mockito.when;

import f2r.tutor_solution.private_tutor_solution_back.config.security.JwtTokenProvider;
import f2r.tutor_solution.private_tutor_solution_back.domain.easy.dto.CollectionParsedRow;
import f2r.tutor_solution.private_tutor_solution_back.domain.easy.entity.EasyBusiness;
import f2r.tutor_solution.private_tutor_solution_back.domain.easy.repository.EasyBusinessRepository;
import f2r.tutor_solution.private_tutor_solution_back.domain.easy.repository.EasyTransactionRepository;
import f2r.tutor_solution.private_tutor_solution_back.domain.easy.service.EasyCollectionIngestService;
import f2r.tutor_solution.private_tutor_solution_back.domain.easy.service.EasyTutorAccessService;
import f2r.tutor_solution.private_tutor_solution_back.domain.easy.service.EasyTutorImportService;
import java.time.LocalDate;
import java.util.List;
import java.util.Map;
import java.util.UUID;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.test.context.bean.override.mockito.MockitoBean;
import org.springframework.transaction.annotation.Transactional;

/** Claude 독립 검수 프로브(스크래치): 강사(TUTOR)가 허용되지 않은 사업체의 수집 배치를 읽거나 확정/분류할 수 있는가. */
@SpringBootTest
@Transactional
class IndependentFinanceProbeMySqlTest {
  @Autowired EasyCollectionIngestService ingest;
  @Autowired EasyTutorImportService tutorImport;
  @Autowired EasyBusinessRepository businesses;
  @Autowired EasyTransactionRepository transactions;
  @MockitoBean EasyTutorAccessService access;

  private long ledger(Long businessId) {
    return transactions.findAll().stream().filter(t -> t.getBusiness().getBusinessId().equals(businessId)).count();
  }

  @Test
  void tutorOfBusinessAcannotReadConfirmOrClassifyBatchOfBusinessB() {
    // given: 사업체 A(강사 허용)와 B(허용 아님). B에 수집 초안 배치 생성
    EasyBusiness a = businesses.findAll().stream().filter(EasyBusiness::isActive).findFirst().orElseThrow();
    EasyBusiness b = businesses.save(EasyBusiness.builder().code("PROBE-" + UUID.randomUUID().toString().substring(0, 8)).name("프로브B").sortOrder(99).build());
    org.springframework.test.util.ReflectionTestUtils.setField(b, "active", true);
    b = businesses.saveAndFlush(b);
    final EasyBusiness bb = b;
    when(access.requireAllowedBusiness(any(), eq(a.getBusinessId()))).thenReturn(a);
    when(access.requireAllowedBusiness(any(), eq(bb.getBusinessId()))).thenThrow(new SecurityException("허용되지 않은 사업체"));
    var row = new CollectionParsedRow(1, Map.of("n", "x"), "{\"n\":\"x\"}", LocalDate.now(), "EXPENSE", 7_777L, "프로브", "타 사업체 접근", "TRANSFER", null, null, "NONE", "프로브계좌", "ext-" + UUID.randomUUID(), null);
    Long batchOfB = ingest.ingestFinancialDraft(bb.getBusinessId(), UUID.randomUUID().toString(), "FIN_" + "b".repeat(24), List.of(row)).run().importBatchId();
    long beforeA = ledger(a.getBusinessId()), beforeB = ledger(bb.getBusinessId());
    var tutor = new JwtTokenProvider.AuthenticatedPrincipal(10L, "tutor-a", "TUTOR");
    // when / then 1) 사업체 B를 직접 지정: 접근 거부
    assertThatThrownBy(() -> tutorImport.detail(tutor, bb.getBusinessId(), batchOfB)).isInstanceOf(RuntimeException.class);
    assertThatThrownBy(() -> tutorImport.confirm(tutor, bb.getBusinessId(), batchOfB, null)).isInstanceOf(RuntimeException.class);
    // 2) 사업체 A로 위장하고 B의 배치 ID를 넣기: 소속 불일치로 거부
    assertThatThrownBy(() -> tutorImport.detail(tutor, a.getBusinessId(), batchOfB)).isInstanceOf(IllegalArgumentException.class).hasMessageContaining("사업체에 속한");
    assertThatThrownBy(() -> tutorImport.confirm(tutor, a.getBusinessId(), batchOfB, null)).isInstanceOf(IllegalArgumentException.class);
    assertThatThrownBy(() -> tutorImport.selectAccount(tutor, a.getBusinessId(), batchOfB, 1L, 1L)).isInstanceOf(IllegalArgumentException.class);
    assertThatThrownBy(() -> tutorImport.reclassify(tutor, a.getBusinessId(), batchOfB, null)).isInstanceOf(IllegalArgumentException.class);
    assertThatThrownBy(() -> tutorImport.discard(tutor, a.getBusinessId(), batchOfB)).isInstanceOf(IllegalArgumentException.class);
    // 3) 어느 쪽 장부도 변하지 않음
    assertThat(ledger(a.getBusinessId())).isEqualTo(beforeA);
    assertThat(ledger(bb.getBusinessId())).isEqualTo(beforeB);
  }
}
